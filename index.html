async function prepare() { const source = document.getElementById("source").value; const result = document.getElementById("result"); if (!source.trim()) { result.textContent = "Please enter Lua source first."; return; } result.textContent = "Analyzing source..."; try { const response = await fetch("/api/analyze", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ source }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Analysis failed"); } result.textContent = "ZERQX ANALYSIS COMPLETE\n\n" + "Language: " + data.language + "\n" + "Lines: " + data.statistics.lines + "\n" + "Characters: " + data.statistics.characters + "\n\n" + "Warnings:\n" + (data.warnings.length ? data.warnings.join("\n") : "No flagged patterns found."); } catch (error) { result.textContent = "ERROR: " + error.message; } } function clearAll() { document.getElementById("source").value = ""; document.getElementById("result").textContent = "Your analysis will appear here."; } async function copyOutput() { const text = document.getElementById("result").textContent; await navigator.clipboard.writeText(text); }
